\noindent
\begin{figure}[H]
	\includegraphics[width=\textwidth,height=0.85\textheight,keepaspectratio]{FigS1_alpha.country.pdf} 
	\caption{Adult butterfly diversity differs by host taxonomy, but not among countries
		(\textbf{a}) Shannon Index differs significantly among host subfamilies (colors) and host tribes (grouped), (\textbf{b}) but not among countries. The full model explained 32\%{} of variation in Shannon diversity, with the main effect accounted to host tribe  ($R^2_{\mathrm{partial}} = 0.24,\; F_{11,154} = 4.5,\; p < 10^{-5}$), there was no effect of country ($R^2_{\mathrm{partial}} < 0.01,\; F_{1,154} = 0.01,\; p = 0.91$). (\textbf{c}) Shannon Index does not differ by country when testing different subfamilies individually: Coliadinae ($p = 0.33$), Dismorphiinae ($p = 0.18$) and Pierinae ($p = 0.91$).}
\end{figure}

\noindent
\begin{figure}[H]
\includegraphics[width=\textwidth,height=0.85\textheight,keepaspectratio]{FigS2_alphaelevation.pdf} 
\caption{Influence of environmental gradients on adult butterfly microbiota diversity
(\textbf{a}) Alpha diversity (only Peru samples) shows linear decrease with elevation (meters above sea level). (\textbf{b}) Alpha diversity (only Germany samples) shows linear increase by calendar week, when removing \textit{Aglais} as outgroup. (\textbf{c}) Shannon Index shows an increase with mean sample week temperature on the complete dataset. (\textbf{d}) Microbial richness increases with mean sample week temperature on the complete dataset. The full model with temperature and host subfamily explained 57\%{} of the observed variation ($F_{13,153} = 17.66, p < 10^{-15}$). Most of the variation was explained by host subfamily turnover ($R^2_{\mathrm{partial}} = 0.33, F_{6,153} = 12.65, p < 10^{-10}$), while temperature had a smaller but significant effect ($R^2_{\mathrm{partial}} = 0.04, F_{1,153} = 6.46, p = 0.012$), the interaction between host subfamily and temperature was also significant ($R^2_{\mathrm{partial}} = 0.19, F_{6,153} = 5.98, p < 0.0001$).}
\end{figure}

\noindent
\begin{figure}[H]
\includegraphics[width=\textwidth,height=0.85\textheight,keepaspectratio]{FigS4_corevenn_alternative.pdf}
\caption{Selection of genus core microbiota
(\textbf{a}) Heatmap highlighting the most abundant and prevalent genera among all samples. (\textbf{b}) Venn diagram shows overlap of core genera when applying different core definitions. High abundance definition (5\% rel. abundance 10\% prevalence) results in 11 core genera (missing \textit{Acinetobacter}); Medium abundance definition (1\% rel. abundance 20\% prevalence) results in 12 core genera (as used in this study). Low abundance definition (0.2\% rel. abundance 40\% prevalence) results in 12 core genera replacing \textit{Entomomonas} and \textit{Apibacter} with \textit{Carnobacterium} and \textit{Raoultella}. (\textbf{c}) Average relative abundance of the 12 genus core taxa among individual butterfly genera. (\textbf{d}) Remaining non-core microbiota among individual butterfly genera. Even non-core genera belonged to the same bacterial groups e.g. LAB (\textit{Carnobacterium}, \textit{Leuconostoc}), AAB (\textit{Swaminathania}, \textit{Saccharibacter}), enteric bacteria (\textit{Cedecea}, \textit{Serratia}, \textit{Roultella}), or Orbacea (\textit{Frischella}). \textit{Aglais} served as an outgroup within the dataset.}
\end{figure}
